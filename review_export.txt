def get_spot_score_and_genre(weather_data, spot_info):
    """為景點計算出景指數和最佳題材"""
    try:
        if not weather_data or not weather_data.get('hourly'):
            return 30, '無資料', '資料不足'
        
        current_time = datetime.now()
        h = weather_data['hourly']
        times = h['time']
        
        # 找到最接近當前時間的索引
        current_index = 0
        for i, time_str in enumerate(times):
            dt = datetime.fromisoformat(time_str.replace('T', ' '))
            if dt <= current_time:
                current_index = i
            else:
                break
        
        # 確保索引有效
        if current_index >= len(times):
            current_index = len(times) - 1
            
        # 獲取當前天氣數據
        current_weather = {}
        for key in h:
            if h[key] and len(h[key]) > current_index:
                current_weather[key] = h[key][current_index]
        
        # 添加景點資訊
        current_weather.update({
            'elevation': weather_data.get('elevation', 0),
            'spot_name': spot_info.get('name', ''),
            'lat': spot_info.get('lat', 0),
            'lon': spot_info.get('lon', 0)
        })
        
        # 調用分類器
        scene_result = classify_weather_scene(current_weather)
        best_genre = scene_result.get('best_genre', '風景攝影')
        score = scene_result.get('score', 30)
        
        return score, best_genre, '天氣分析完成'
        
    except Exception as e:
        return 30, '風景攝影', f'計算錯誤: {str(e)}'

def summary():
    region = request.args.get('region', 'tw')
    sub_region = request.args.get('sub', '')
    
    # 檢查是否需要更新快取
    global last_cache_update
    now = datetime.now()
    cache_key = f"{region}_{sub_region}" if sub_region else region
    
    if (cache_key not in last_cache_update or 
        now - last_cache_update[cache_key] > timedelta(hours=6)):
        
        print(f"Updating cache for {cache_key}...")
        update_cache_if_needed(region)
        last_cache_update[cache_key] = now
    
    try:
        spots = get_region_spots(region, sub_region)
        results = []
        
        for spot in spots:
            weather_data = get_cached_weather(spot['lat'], spot['lon'])
            if weather_data:
                score, genre, status = get_spot_score_and_genre(weather_data, spot)
                
                # 獲取天氣提示
                weather_tip = get_weather_based_tip(weather_data, spot, region)
                
                spot_result = {
                    'name': spot['name'],
                    'lat': spot['lat'],
                    'lon': spot['lon'],
                    'elevation': weather_data.get('elevation', spot.get('elevation', 0)),
                    'today': {
                        'score': int(score),
                        'best_genre': genre,
                        'status': status,
                        'weather_tip': weather_tip
                    }
                }
                
                # 嘗試獲取未來幾天的預測
                if weather_data.get('hourly', {}).get('time'):
                    try:
                        # 簡化的未來預測（使用相同邏輯）
                        spot_result['tomorrow'] = {'score': max(10, int(score * 0.9)), 'best_genre': genre}
                        spot_result['day_after'] = {'score': max(10, int(score * 0.8)), 'best_genre': genre}
                    except:
                        spot_result['tomorrow'] = {'score': 30, 'best_genre': '風景攝影'}
                        spot_result['day_after'] = {'score': 30, 'best_genre': '風景攝影'}
                
                results.append(spot_result)
            else:
                # 無天氣資料時的預設值
                results.append({
                    'name': spot['name'],
                    'lat': spot['lat'], 
                    'lon': spot['lon'],
                    'elevation': spot.get('elevation', 0),
                    'today': {'score': 30, 'best_genre': '風景攝影', 'status': '無天氣資料'},
                    'tomorrow': {'score': 30, 'best_genre': '風景攝影'},
                    'day_after': {'score': 30, 'best_genre': '風景攝影'}
                })
    
        # 按評分排序
        results.sort(key=lambda x: x['today']['score'], reverse=True)
        
        # 準備模板資料
        template_data = {
            'spots': results,
            'region': region,
            'sub_region': sub_region,
            'region_name': {
                'tw': '台灣', 'jp': '日本', 'us': '美國', 'ak': '阿拉斯加'
            }.get(region, region.upper()),
            'sub_regions': get_sub_regions_by_key(region),
            'update_time': now.strftime('%Y-%m-%d %H:%M')
        }
        
        return render_template('summary.html', **template_data)
        
    except Exception as e:
        return render_template('error.html', error=str(e))

CLOUD_LAYERS = {
    'low':  (0, 2000),    # 低雲：0~2,000m
    'mid':  (2000, 6000), # 中雲：2,000~6,000m
    'high': (6000, 12000),# 高雲：6,000m 以上
}

def calc_cloud_base(temp_c, dew_c):
    """使用溫度-露點差估算雲底高度（公尺 AGL）"""
    spread = temp_c - dew_c
    if spread <= 0:
        return 0  # 已飽和，雲底在地面
    base_agl = spread * 125
    return min(base_agl, 6000)  # 合理上限

def classify_photographer_position(elevation, cloud_base_agl, low_cc, mid_cc, high_cc):
    """
    判斷攝影師相對於雲層的位置。
    回傳：position, description, score_modifier
    """
    cloud_base_msl = cloud_base_agl + elevation  # 雲底海拔

    # ── 情況 1：低雲層大量存在 ──
    if low_cc > 40:
        low_ceiling = CLOUD_LAYERS['low'][1]  # 2,000m
        if elevation > low_ceiling:
            # 攝影師在低雲層上方（如合歡山 3,300m > 2,000m）
            return ('above_low',
                    '🏔️ 站在低雲層之上 → 絕佳雲海視角！',
                    1.2)
        elif elevation > cloud_base_msl - 200:
            # 攝影師接近/在雲底高度 → 可能身處霧中
            if low_cc > 70:
                return ('in_cloud',
                        '🌫️ 身處雲層中 → 濃霧，能見度差 ⚠️',
                        0.3)
            else:
                return ('near_cloud',
                        '🌁 接近雲層邊緣 → 局部霧氣',
                        0.7)
        else:
            # 攝影師在低雲下方
            return ('below_low',
                    '☁️ 雲在頭頂上方 → 天氣陰沉',
                    0.5)

    # ── 情況 2：中雲層為主 ──
    if mid_cc > 50:
        if elevation > 3000:
            return ('above_mid',
                    '🏔️ 接近中雲層高度 → 有機會雲海或身處雲中',
                    1.0)
        else:
            return ('below_mid',
                    '☁️ 中雲在頭頂 → 一般多雲天氣',
                    0.6)

    # ── 情況 3：晴朗或高雲 ──
    if high_cc > 50:
        return ('high_only',
                '☀️ 高雲為主 → 火燒雲素材！',
                1.0)

    return ('clear',
            '☀️ 晴朗無雲 → 適合一般風景攝影',
            0.8)

    best_temp = h['temperature_2m'][best_i] or 0
    best_dew = h['dew_point_2m'][best_i] or 0
    best_cl = h['cloud_cover_low'][best_i] or 0
    best_cm = h['cloud_cover_mid'][best_i] or 0
    best_ws = h['wind_speed_10m'][best_i] or 0
    best_vis = h['visibility'][best_i] or 0
    best_wc = h['weather_code'][best_i]
    best_wc_desc = WMO_DESC.get(best_wc, '') if best_wc is not None else ''
    best_cb = calc_cloud_base(best_temp, best_dew)

    print(f'  🏆 最佳：{best_t}（{best_sc}分）')
    print(f'     📋 {best_reason}')
    print(f'     📐 海拔{elevation:.0f}m | 雲底{best_cb:.0f}m AGL ({best_cb+elevation:.0f}m MSL)')
    print(f'     🌡️ {best_temp:.0f}°C 露點{best_dew:.0f}°C | 低雲{best_cl}% 中雲{best_cm}% | 風{best_ws:.1f}km/h 能見{best_vis//1000}km | {best_wc_desc}')

    # 專業攝影建議
    print(f'     📷 鏡頭建議：', end='')
    if is_mt:
        if best_cl > 60 and elevation > 2000:
            print('14-24mm超廣角 + ND減光鏡（雲海縮時）')
            print(f'     ⚠️ 雲海在腳下{best_cb:.0f}m處，三腳架要夠穩，風{best_ws:.1f}km/h')
        elif best_cl > 50:
            print('24-70mm標準變焦 + 70-200mm長焦（雲海層次壓縮）')
        else:
            print('16-35mm廣角（星空銀河），ISO 3200-6400')
    else:
        print('70-200mm長焦（壓縮夕陽/倒影），漸層減光鏡')

print(f'\n{"=" * 95}')
print('🌤️  新增：雲底高度估算（T-Td）×125，判斷拍照者與雲層的垂直位置')
print('📐  WMO雲層標準：低雲 0~2km | 中雲 2~6km | 高雲 >6km')
print('💡  判斷邏輯：海拔 > 雲層頂 → 雲海上空 | 海拔 ≈ 雲底 → 身處霧中')

// 地圖初始化
function initMap() {
  // 初始化地圖（以台灣為中心）
  photoMap = L.map('chaselights-map').setView([23.8, 121.0], 6);
  
  // 使用 OpenStreetMap 免費圖磚
  L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
    maxZoom: 18,
    attribution: '© OpenStreetMap contributors'
  }).addTo(photoMap);
  
  // 載入景點資料
  loadAllSpots();
}

// 載入所有區域的景點資料
async function loadAllSpots() {
  const regions = ['tw', 'jp', 'us', 'ak'];
  
  for (const region of regions) {
    try {
      const response = await fetch(`/api/best-spots?region=${region}`);
      const data = await response.json();
      
      if (data && data.today) {
        // 為每個景點添加標記
        data.today.forEach(spot => {
          if (spot.lat && spot.lon) {
            addSpotMarker(spot, region);
          }
        });
      }
    } catch (e) {
      console.warn(`Failed to load ${region} spots:`, e);
    }
  }
}

// 添加景點標記
function addSpotMarker(spot, region) {
  const score = spot.today?.score || 0;
  const color = getMarkerColor(score);
  
  // 創建客製化標記
  const marker = L.circleMarker([spot.lat, spot.lon], {
    radius: 8,
    fillColor: color,
    color: '#fff',
    weight: 2,
    opacity: 1,
    fillOpacity: 0.8
  }).addTo(photoMap);
  
  // 創建彈出視窗
  const popupContent = `
    <div class="map-popup">
      <div class="popup-title">${spot.name}</div>
      <div class="popup-score">
        <span class="score-label">${currentLang === 'zh' ? '出景指數' : 'Score'}</span>
        <span class="score-value" style="color:${color}">${score}</span>
      </div>
      <div class="popup-genre">${spot.today?.best_genre || 'N/A'}</div>
      <a href="/summary?region=${region}" class="popup-link">
        ${currentLang === 'zh' ? '📋 查看詳細' : '📋 View Details'}
      </a>
    </div>
  `;
  
  marker.bindPopup(popupContent);
}

// 根據評分獲取標記顏色
function getMarkerColor(score) {
  if (score >= 80) return '#00c853';
  if (score >= 60) return '#64dd17';
  if (score >= 40) return '#ffd600';
  if (score >= 20) return '#ff6d00';
  return '#d50000';
}

}
